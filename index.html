Orders Data SQL Query 
SELECT o.userid, o.orderID, o.delCompanyAddress, o.delCompanyAddress2,
       o.delLandmark, o.delPincode, o.delCity, o.delState,
       CONCAT(ol.latitude,',',ol.longitude) as deliveryLatLong,
       ol.createdOn as oDate, sm.fmWarehouseId as dsid,
       pu.userName as deliveryAgent,
       JSON_UNQUOTE(ol.requestData->'$.podImg[0]') as podImg
FROM orders o
INNER JOIN open_orders_log ol On ol.orderId=o.orderID
INNER JOIN seller_master sm ON sm.userid = o.sellerID
LEFT JOIN panel_users pu ON pu.userId=ol.createdBy
WHERE o.delAddressId=1
  AND DATE_FORMAT(o.updated_at,'%Y%m%d')>20251005
ORDER BY o.userId
Darkstores Data SQL Query 
SELECT lpa.uniqueId as dsid,
       CONCAT(lpa.latitude,',',lpa.longitude) as DarkStoreLatLong
FROM lp_address lpa
WHERE lpa.warehouseType=2
